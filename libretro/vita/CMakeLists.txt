cmake_minimum_required(VERSION 3.20)
project(pix68k_libretro_portable CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(PIX68K_ROOT "${CMAKE_CURRENT_LIST_DIR}/../..")
set(PIX68K_BUILD_CPU020 OFF)
set(PIX68K_BUILD_FPU_BOARD OFF)
include("${PIX68K_ROOT}/cmake/emulation-core.cmake")
option(PIX68K_PORTABLE_TEST_SHARED "Build a shared reduced-profile core for native validation" OFF)
if(PIX68K_PORTABLE_TEST_SHARED)
  if(CMAKE_CROSSCOMPILING)
    message(FATAL_ERROR "Vita requires static linking")
  endif()
  set(core_kind SHARED)
  set_target_properties(m68k x68k PROPERTIES POSITION_INDEPENDENT_CODE ON)
else()
  if(NOT CMAKE_CROSSCOMPILING)
    message(FATAL_ERROR "Use the VitaSDK toolchain, or PIX68K_PORTABLE_TEST_SHARED=ON for native validation")
  endif()
  set(core_kind STATIC)
  include(CheckIPOSupported)
  check_ipo_supported(RESULT vita_ipo_supported OUTPUT vita_ipo_error LANGUAGES CXX)
  if(NOT vita_ipo_supported)
    message(FATAL_ERROR "Vita build requires GCC LTO support: ${vita_ipo_error}")
  endif()
  if(NOT CMAKE_CXX_COMPILER_AR)
    message(FATAL_ERROR "Vita build requires the SDK's plugin-aware gcc-ar")
  endif()
  set_property(TARGET m68k x68k PROPERTY INTERPROCEDURAL_OPTIMIZATION TRUE)
  # The SDK's generic ARMv7 tuning leaves measurable performance on Cortex-A9.
  target_compile_options(m68k PRIVATE -mcpu=cortex-a9)
  target_compile_options(x68k PRIVATE -mcpu=cortex-a9)
endif()
add_library(pix68k_libretro ${core_kind} "${PIX68K_ROOT}/libretro/core.cpp"
  "${PIX68K_ROOT}/libretro/media.cpp" "${PIX68K_ROOT}/libretro/state.cpp" "${PIX68K_ROOT}/libretro/render_worker.cpp" "${PIX68K_ROOT}/libretro/audio_worker.cpp")
target_compile_definitions(pix68k_libretro PRIVATE PIX68K_VITA)
target_include_directories(pix68k_libretro PRIVATE "${PIX68K_ROOT}/external/libretro")
target_compile_options(pix68k_libretro PRIVATE -Wall -Wextra -fno-exceptions -fno-rtti)
option(PIX68K_VITA_GPU_LAYERS "Build palette/layer GPU bridge for the packaged vita2d frontend" ON)
if(NOT PIX68K_PORTABLE_TEST_SHARED)
  if(PIX68K_VITA_GPU_LAYERS)
    target_sources(pix68k_libretro PRIVATE "${PIX68K_ROOT}/libretro/vita/paletted_gpu.cpp")
    target_compile_definitions(pix68k_libretro PRIVATE PIX68K_VITA_GPU_LAYERS)
  endif()
  target_compile_options(pix68k_libretro PRIVATE -mcpu=cortex-a9)
  set_property(TARGET pix68k_libretro PROPERTY INTERPROCEDURAL_OPTIMIZATION TRUE)
endif()
target_link_libraries(pix68k_libretro PRIVATE x68k)
if(PIX68K_PORTABLE_TEST_SHARED)
  find_package(Threads REQUIRED)
  target_link_libraries(pix68k_libretro PRIVATE Threads::Threads)
  set_target_properties(pix68k_libretro PROPERTIES PREFIX "")
  enable_testing()
  add_executable(vita_profile_audio_worker_test "${PIX68K_ROOT}/tests/libretro/audio_worker_test.cpp" "${PIX68K_ROOT}/libretro/audio_worker.cpp")
  target_include_directories(vita_profile_audio_worker_test PRIVATE "${PIX68K_ROOT}/libretro" "${PIX68K_ROOT}/tests/unit")
  target_compile_options(vita_profile_audio_worker_test PRIVATE -fno-exceptions -fno-rtti)
  target_link_libraries(vita_profile_audio_worker_test PRIVATE x68k Threads::Threads)
  add_test(NAME vita_profile_audio_worker COMMAND vita_profile_audio_worker_test)
  set_tests_properties(vita_profile_audio_worker PROPERTIES TIMEOUT 30)
  add_executable(vita_profile_render_worker_test "${PIX68K_ROOT}/tests/libretro/render_worker_test.cpp" "${PIX68K_ROOT}/libretro/render_worker.cpp")
  target_include_directories(vita_profile_render_worker_test PRIVATE "${PIX68K_ROOT}/libretro" "${PIX68K_ROOT}/tests/unit")
  target_compile_options(vita_profile_render_worker_test PRIVATE -fno-exceptions -fno-rtti)
  target_link_libraries(vita_profile_render_worker_test PRIVATE x68k Threads::Threads)
  add_test(NAME vita_profile_render_worker COMMAND vita_profile_render_worker_test)
  set_tests_properties(vita_profile_render_worker PROPERTIES TIMEOUT 30)
  add_executable(vita_profile_media_test "${PIX68K_ROOT}/tests/libretro/media_test.cpp"
    "${PIX68K_ROOT}/libretro/media.cpp")
  target_compile_definitions(vita_profile_media_test PRIVATE PIX68K_VITA)
  target_include_directories(vita_profile_media_test PRIVATE "${PIX68K_ROOT}/libretro" "${PIX68K_ROOT}/tests/unit")
  target_link_libraries(vita_profile_media_test x68k)
  add_test(NAME vita_profile_media COMMAND vita_profile_media_test)
  add_executable(vita_profile_api_test "${PIX68K_ROOT}/tests/libretro/api_test.cpp")
  target_include_directories(vita_profile_api_test PRIVATE "${PIX68K_ROOT}/external/libretro" "${PIX68K_ROOT}/tests/unit")
  target_link_libraries(vita_profile_api_test PRIVATE ${CMAKE_DL_LIBS})
  add_dependencies(vita_profile_api_test pix68k_libretro)
  add_test(NAME vita_profile_api COMMAND vita_profile_api_test $<TARGET_FILE:pix68k_libretro>)
  add_executable(vita_profile_runtime_test "${PIX68K_ROOT}/tests/libretro/runtime_test.cpp")
  target_include_directories(vita_profile_runtime_test PRIVATE "${PIX68K_ROOT}/external/libretro" "${PIX68K_ROOT}/tests/unit")
  target_link_libraries(vita_profile_runtime_test PRIVATE x68k ${CMAKE_DL_LIBS})
  add_dependencies(vita_profile_runtime_test pix68k_libretro)
  add_test(NAME vita_profile_runtime COMMAND vita_profile_runtime_test $<TARGET_FILE:pix68k_libretro>)
  add_test(NAME vita_profile_human68k COMMAND vita_profile_runtime_test $<TARGET_FILE:pix68k_libretro>
    "$ENV{HOME}/.cache/pix68k/rom" "$ENV{HOME}/.cache/pix68k/disk/HUMAN302.XDF")
  set_tests_properties(vita_profile_runtime PROPERTIES TIMEOUT 60)
  set_tests_properties(vita_profile_human68k PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 120)
  return()
endif()
# RetroArch's Vita build consumes one archive, including all emulator objects.
add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/libretro_vita.a"
  COMMAND ${CMAKE_COMMAND} -DARCHIVER=${CMAKE_CXX_COMPILER_AR}
    -DINPUT=${CMAKE_CURRENT_BINARY_DIR}/combine.mri -P "${CMAKE_CURRENT_LIST_DIR}/combine.cmake"
  DEPENDS pix68k_libretro x68k m68k
    "$<TARGET_FILE:pix68k_libretro>" "$<TARGET_FILE:x68k>" "$<TARGET_FILE:m68k>"
    "${CMAKE_CURRENT_BINARY_DIR}/combine.mri"
    "${CMAKE_CURRENT_LIST_DIR}/combine.cmake"
  VERBATIM)
add_custom_target(vita_core_archive ALL DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/libretro_vita.a")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/combine.mri" CONTENT
"CREATE libretro_vita.a\nADDLIB $<TARGET_FILE_NAME:pix68k_libretro>\nADDLIB $<TARGET_FILE_NAME:x68k>\nADDLIB $<TARGET_FILE_NAME:m68k>\nSAVE\nEND\n")
